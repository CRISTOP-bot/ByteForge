#!/usr/bin/env python3
"""Train ByteForge's portable token n-gram model from scratch."""
import argparse
import re
from collections import Counter, defaultdict
from pathlib import Path

ROOT = Path(__file__).resolve().parent
TOKEN_RE = re.compile(r"[\w]+|[^\w\s]", re.UNICODE)


def train(text: str, order: int = 2) -> str:
    if order < 1:
        raise ValueError("order must be at least 1")
    tokens = TOKEN_RE.findall(text)
    if not tokens:
        raise ValueError("training corpus is empty")
    vocab = sorted(set(tokens))
    ids = {token: i for i, token in enumerate(vocab)}
    counts = Counter(tuple(ids[t] for t in tokens[max(0, i-order):i]) + (ids[tokens[i]],)
                     for i in range(len(tokens)))
    # Include all contexts in fixed-width order; left-pad sentence start with <BOS>.
    bos = len(vocab)
    transitions = Counter()
    history = [bos] * order
    for token in tokens:
        nxt = ids[token]
        transitions[tuple(history) + (nxt,)] += 1
        history = history[1:] + [nxt]
    lines = [f"BYTEFORGE_NGRAM_V1\t{order}\t{len(vocab)}\t{bos}"]
    lines += [f"v\t{i}\t{token.encode('utf-8').hex()}" for i, token in enumerate(vocab)]
    lines += ["n\t" + "\t".join(map(str, key)) + f"\t{count}"
              for key, count in sorted(transitions.items())]
    return "\n".join(lines) + "\n"


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--corpus", type=Path, default=ROOT / "data/corpus.txt")
    parser.add_argument("--output", type=Path, default=ROOT / "model.bfng")
    parser.add_argument("--order", type=int, default=2)
    args = parser.parse_args()
    model = train(args.corpus.read_text(encoding="utf-8"), args.order)
    args.output.write_text(model, encoding="ascii")
    print(f"ByteForge entrenado: {sum(1 for line in model.splitlines() if line.startswith('n\t'))} transiciones -> {args.output}")


if __name__ == "__main__":
    main()
